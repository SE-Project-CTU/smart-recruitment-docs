% --- CẤU HÌNH LỀ TRANG ---
\usepackage[a4paper,
    top=2.5cm,
    bottom=2.5cm,
    left=2.5cm,
    right=2.0cm,
    headheight=20pt
]{geometry}

\usepackage{tabularx}
\usepackage{array}

% --- ĐỊNH NGHĨA CÁC KIỂU CỘT CĂN LỀ TOÀN CỤC ---
% 1. Cột X tự động căn trái (thay thế cho X mặc định khi cần)
\newcolumntype{Y}{>{\raggedright\arraybackslash}X}

% 2. Các cột p có kích thước cố định (không bị giãn chữ)
\newcolumntype{L}[1]{>{\raggedright\arraybackslash}p{#1}} % Căn trái
\newcolumntype{C}[1]{>{\centering\arraybackslash}p{#1}}   % Căn giữa
\newcolumntype{R}[1]{>{\raggedleft\arraybackslash}p{#1}}  % Căn phải

% 3. Cấu hình lại cột X gốc tự động căn trái toàn bộ dự án
\renewcommand{\tabularxcolumn}[1]{>{\raggedright\arraybackslash}p{#1}}

\usepackage{etoolbox}

% Thiết lập khoảng cách 0.5cm phía trên và phía dưới cho tất cả môi trường table / tabularx
\AtBeginEnvironment{tabularx}{\vspace{0.5cm}}
\AtEndEnvironment{tabularx}{\vspace{0.5cm}}

% (Tùy chọn) Nếu bạn dùng môi trường float \begin{table}...
\setlength{\intextsep}{0.5cm plus 2pt minus 2pt} % Khoảng cách cho bảng nằm giữa văn bản


% --- CẤU HÌNH HEADER & FOOTER (TIMES NEW ROMAN) ---
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{} % Xóa cấu hình mặc định

% Tạo định nghĩa kiểu chữ Times New Roman cho Header/Footer
\newcommand{\headerfont}{\small\rmfamily\itshape}

% Áp dụng cho trang thường
\fancyhead[L]{\small\fontspec{Times New Roman}\itshape\bfseries Tài liệu thiết kế phần mềm}
\fancyhead[R]{\small\fontspec{Times New Roman}\itshape\bfseries Trang \thepage}

\fancyfoot[L]{\small\fontspec{Times New Roman}\itshape\bfseries Bộ môn CNPM, Khoa CNTT \& TT, Đại học Cần Thơ}
\fancyfoot[R]{}

% Xóa đường kẻ ngang header & footer
\renewcommand{\headrulewidth}{0pt}
\renewcommand{\footrulewidth}{0pt}

% Đồng bộ cấu hình cho trang 'plain' (trang đầu chương, trang mục lục)
\fancypagestyle{plain}{
    \fancyhf{}
    \fancyhead[L]{\small\fontspec{Times New Roman}\itshape\bfseries Tài liệu thiết kế phần mềm}
    \fancyhead[R]{\small\fontspec{Times New Roman}\itshape\bfseries Trang \thepage}
    \fancyfoot[L]{\small\fontspec{Times New Roman}\itshape\bfseries Bộ môn CNPM, Khoa CNTT \& TT, Đại học Cần Thơ}
    \fancyfoot[R]{}
    \renewcommand{\headrulewidth}{0pt}
    \renewcommand{\footrulewidth}{0pt}
}
