% --- CẤU HÌNH LỀ TRANG ---
\usepackage[a4paper,
    top=1.5cm,
    bottom=1.5cm,
    left=3.5cm,
    right=2.0cm,
    headheight=16pt,  % Độ cao khung chứa header
    headsep=0.4cm,    % Khoảng cách giữa đường kẻ Header và dòng LỜI CẢM ƠN
    footskip=1cm,   % Khoảng cách giữa dòng cuối văn bản và Footer
    includehead,      % ĐẢM BẢO: Header nằm trọn bên trong khoảng lề 2.5cm phía trên
    includefoot       % ĐẢM BẢO: Footer nằm trọn bên trong khoảng lề 2.5cm phía dưới
]{geometry}

\usepackage{tabularx}
\usepackage{array}

% --- ĐỊNH NGHĨA CÁC KIỂU CỘT CĂN LỀ TOÀN CỤC ---
% 1. Cột X tự động căn trái (thay thế cho X mặc định khi cần)
\newcolumntype{Y}{>{\raggedright\arraybackslash}X}

% 2. Các cột p có kích thước cố định (không bị giãn chữ)
\newcolumntype{L}[1]{>{\raggedright\arraybackslash}p{#1}} % Căn trái
\newcolumntype{C}[1]{>{\centering\arraybackslash}p{#1}}   % Căn giữa
\newcolumntype{R}[1]{>{\raggedleft\arraybackslash}p{#1}}  % Căn phải

% 3. Cấu hình lại cột X gốc tự động căn trái toàn bộ dự án
\renewcommand{\tabularxcolumn}[1]{>{\raggedright\arraybackslash}p{#1}}

\usepackage{etoolbox}

% Thiết lập khoảng cách 0.5cm phía trên và phía dưới cho tất cả môi trường table / tabularx
\AtBeginEnvironment{tabularx}{\vspace{0.5cm}}
\AtEndEnvironment{tabularx}{\vspace{0.5cm}}

% (Tùy chọn) Nếu bạn dùng môi trường float \begin{table}...
\setlength{\intextsep}{0.5cm plus 2pt minus 2pt} % Khoảng cách cho bảng nằm giữa văn bản


\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{} % Xóa cấu hình mặc định

% --- CẤU HÌNH HEADER (ĐẦU TRANG) ---
\fancyhead[L]{\small\fontspec{Times New Roman}\itshape Khoa Công nghệ Phần mềm}
\fancyhead[R]{} % Hoặc để tên tài liệu/mục nếu muốn

% --- CẤU HÌNH FOOTER (CHÂN TRANG) ---
\fancyfoot[L]{\small\fontspec{Times New Roman}\itshape Trường Công nghệ Thông tin \& Truyền thông}
\fancyfoot[R]{\small\fontspec{Times New Roman}\itshape Trang \thepage}

% --- ĐƯỜNG KẺ NGANG PHÂN CÁCH (RULE) ---
\renewcommand{\headrulewidth}{0.4pt} % Đường kẻ ngang đường đệm Header
\renewcommand{\footrulewidth}{0.4pt} % Đường kẻ ngang đường đệm Footer

% --- ĐỒNG BỘ CHO TRANG 'PLAIN' (Trang đầu chương, Mục lục,...) ---
\fancypagestyle{plain}{
    \fancyhf{}
    \fancyhead[L]{\small\fontspec{Times New Roman}\itshape Khoa Công nghệ Phần mềm}
    \fancyhead[R]{}
    \fancyfoot[L]{\small\fontspec{Times New Roman}\itshape Trường Công nghệ Thông tin \& Truyền thông}
    \fancyfoot[R]{\small\fontspec{Times New Roman}\itshape Trang \thepage}
    
    \renewcommand{\headrulewidth}{0.4pt}
    \renewcommand{\footrulewidth}{0.4pt}
}
