%! TEX root = ../main.tex

\usepackage{titlesec}

% 0. CẤU HÌNH ĐỘ SÂU ĐÁNH SỐ (Đến Heading 4 / \paragraph)
\setcounter{secnumdepth}{4}

% 1. ĐỊNH NGHĨA SỐ THỨ TỰ HEADING
\renewcommand{\thesection}{\arabic{section}}                       % 1, 2, 3...
\renewcommand{\thesubsection}{\thesection.\arabic{subsection}}     % 1.1, 1.2...
\renewcommand{\thesubsubsection}{\thesubsection.\arabic{subsubsection}} % 1.1.1, 1.1.2...
\renewcommand{\theparagraph}{\alph{paragraph})}                     % a), b), c)...

% Đặt lại bộ đếm khi sang chương/mục mới
\makeatletter
\@addtoreset{subsection}{section}
\@addtoreset{paragraph}{subsubsection}
\makeatother

% 2. TÙY CHỈNH KIỂU CHỮ NGOÀI TRANG
% Heading 1 (\section) -> CHƯƠNG 1: TIÊU ĐỀ
\titleformat{\section}[block]
  {\centering\Large\bfseries}
  {CHƯƠNG \thesection:}
  {0.5em}
  {\uppercase}

% Heading 2 (\subsection) -> 1.1 Tiêu đề
\titleformat{\subsection}
  {\large\bfseries}
  {\thesubsection}
  {0.5em}
  {}

% Heading 3 (\subsubsection) -> 1.1.1 Tiêu đề
\titleformat{\subsubsection}
  {\large\bfseries}
  {\thesubsubsection}
  {0.5em}
  {}

% Heading 4 (\paragraph) -> a) Tiêu đề
\titleformat{\paragraph}
  {\normalfont\bfseries}
  {\theparagraph}
  {0.5em}
  {}

% 3. CẤU HÌNH KHOẢNG CÁCH CHO HEADING (\titlespacing*{câu_lệnh}{lề_trái}{trên}{dưới})
% Giảm tham số thứ 4 (after-sep) để thu hẹp khoảng cách với dòng kế tiếp
\titlespacing*{\section}
  {0pt}{1.5ex plus 0.5ex minus 0.2ex}{0.2ex plus 0.2ex}

\titlespacing*{\subsection}
  {0pt}{1.2ex plus 0.4ex minus 0.2ex}{0.2ex plus 0.2ex}

\titlespacing*{\subsubsection}
  {0pt}{1.0ex plus 0.3ex minus 0.2ex}{0.2ex plus 0.1ex}

\titlespacing*{\paragraph}
  {0pt}{0.8ex plus 0.2ex minus 0.2ex}{0.2ex plus 0.1ex}

% --- CẤU HÌNH DÀNH CHO HÌNH VÀ BẢNG (Chuyển sang dạng 1.1, 1.2) ---
\renewcommand{\figurename}{Hình}
\renewcommand{\tablename}{Bảng}

\makeatletter
\@addtoreset{figure}{section}
\@addtoreset{table}{section}
\makeatother

\renewcommand{\thefigure}{\thesection.\arabic{figure}}
\renewcommand{\thetable}{\thesection.\arabic{table}}

\usepackage{caption}
\captionsetup[figure]{
    labelfont={bf},
    textfont={normalfont},
    labelsep=colon,
    skip=6pt
}

\captionsetup[table]{
    position=top,
    labelfont={bf},
    textfont={normalfont},
    labelsep=colon,
    skip=6pt,
    justification=centering
}

\titlespacing*{\subsubsection}
  {0pt}{1.8ex plus 1ex minus .2ex}{1.0ex plus .2ex}

% --- CẤU HÌNH MẶC ĐỊNH CHO SECTION* TỰ ĐỘNG VÀO MỤC LỤC ---
\let\oldsection\section
\RenewDocumentCommand{\section}{s o m}{%
  \IfBooleanTF{#1}
    {% Khi dùng \section*{...}
      \phantomsection
      \oldsection*{#3}%
      \addcontentsline{toc}{section}{#3}%
    }
    {% Khi dùng \section{...} thông thường
      \IfValueTF{#2}
        {\oldsection[#2]{#3}}
        {\oldsection{#3}}%
    }%
}
