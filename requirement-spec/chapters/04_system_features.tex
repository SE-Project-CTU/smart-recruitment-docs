\subsubsection{Mô tả và mức độ ưu tiên}

\begin{itemize}
  \item \textbf{Mô tả:} Phân hệ cung cấp các chức năng quản lý vòng đời tài khoản người dùng, bao gồm đăng ký tài khoản, đăng nhập, đăng xuất, quản lý thông tin cá nhân, đổi mật khẩu và duy trì phiên đăng nhập. Hệ thống thực hiện xác thực người dùng dựa trên chuẩn JWT và phân quyền theo vai trò đối với 3 nhóm người dùng: Ứng viên, Nhà tuyển dụng và Quản trị viên.
  \item \textbf{Mức độ ưu tiên:} Cao
  \item \textbf{Đánh giá tiêu chí:} Lợi ích: 9/9 | Chi phí: 3/9 | Rủi ro: 7/9
\end{itemize}

    Người dùng đã đăng nhập chọn chức năng Cập nhật thông tin cá nhân. & Hệ thống kiểm tra dữ liệu đầu vào, cập nhật thông tin cá nhân vào CSDL và thông báo kết quả cho người dùng. \\ \hline

\subsubsection{Các yêu cầu chức năng}

\begin{enumerate}[label=\bfseries REQ-AUTH-\arabic*:, labelwidth=!, labelsep=0.5em, leftmargin=*]
    \item \textbf{Đăng ký tài khoản Ứng viên:} Hệ thống phải cho phép người dùng khách đăng ký tài khoản Ứng viên bằng Email, số điện thoại, Mật khẩu và Họ tên.
    \item \textbf{Đăng ký tài khoản Nhà tuyển dụng:} Hệ thống phải cho phép người dùng khách đăng ký tài khoản Nhà tuyển dụng bằng Email, Mật khẩu, Họ tên và thông tin Tên công ty ban đầu.
    \item \textbf{Xác thực đăng nhập JWT:} Hệ thống phải xác thực người dùng dựa trên Email/Mật khẩu và phát hành JWT Token chứa thông tin định danh và vai trò (Role Claims) sau khi xác thực thành công.
    \item \textbf{Mã hóa mật khẩu:} Hệ thống phải mã hóa tất cả mật khẩu người dùng bằng giải thuật băm an toàn (BCrypt hoặc Argon2) trước khi lưu trữ vào CSDL.
    \item \textbf{Phân quyền người dùng - RBAC:} Hệ thống phải kiểm tra và giới hạn quyền truy cập API/chức năng dựa trên 3 nhóm vai trò: Ứng viên, Nhà tuyển dụng, Quản trị viên.
    \item \textbf{Cấp lại Token / Refresh Token:} Hệ thống phải cung cấp cơ chế Refresh Token cho phép duy trì phiên đăng nhập của người dùng mà không bắt người dùng đăng nhập lại liên tục khi Access Token hết hạn.
    \item \textbf{Đăng xuất và quản lý phiên:} Hệ thống phải cho phép người dùng đăng xuất và vô hiệu hóa Token ở phía Server/Client.
    \item \textbf{Xử lý lỗi xác thực và phân quyền:}
    \begin{itemize}
        \item Nếu Email đã tồn tại khi đăng ký, hệ thống phải trả về thông báo lỗi \textit{\textquotedblleft Email này đã được sử dụng\textquotedblright}.
        \item Nếu thông tin đăng nhập sai, hệ thống phải trả về phản hồi chung \textit{\textquotedblleft Tài khoản hoặc mật khẩu không chính xác\textquotedblright} để đảm bảo tính bảo mật.
        \item Nếu Token hết hạn hoặc không hợp lệ, hệ thống phải trả về mã lỗi 401 Unauthorized.
        \item Nếu người dùng truy cập vào chức năng không thuộc thẩm quyền của vai trò, hệ thống phải trả về mã lỗi 403 Forbidden.
    \end{itemize}
    \item \textbf{Đổi mật khẩu:} Hệ thống phải cho phép người dùng đã đăng nhập thay đổi mật khẩu tài khoản bằng cách cung cấp mật khẩu hiện tại và mật khẩu mới. Hệ thống phải kiểm tra tính hợp lệ của mật khẩu hiện tại, mã hóa mật khẩu mới bằng thuật toán băm an toàn trước khi lưu vào cơ sở dữ liệu và thông báo kết quả cho người dùng.
    \item \textbf{Quản lý thông tin cá nhân:} Hệ thống phải cho phép người dùng đã đăng nhập xem và cập nhật thông tin cá nhân thuộc tài khoản của mình.
\end{enumerate}

\subsubsection{Các yêu cầu chức năng}

\begin{enumerate}[label=\bfseries REQ-CV-\arabic*:, leftmargin=2.5cm]
    \item \textbf{Quản lý và Chọn Template CV:} Hệ thống phải hiển thị danh sách các mẫu CV có sẵn kèm theo hình ảnh xem trước (thumbnail) để ứng viên lựa chọn.
    \item \textbf{Biên soạn Thông tin cá nhân và Mục tiêu:} Hệ thống phải cho phép ứng viên nhập và cập nhật trực tiếp: Họ tên, Chức danh công việc, Email, Số điện thoại, Địa chỉ, Liên kết cá nhân (LinkedIn, GitHub, Portfolio) và Mục tiêu nghề nghiệp.
    \item \textbf{Biên soạn Kinh nghiệm làm việc và Học vấn:} Hệ thống phải cho phép ứng viên thêm, sửa, xóa, sắp xếp thứ tự danh sách Kinh nghiệm làm việc (Tên công ty, Vị trí, Thời gian, Mô tả chi tiết) và Trình độ học vấn (Tên trường, Chuyên ngành, Thời gian, Bằng cấp, GPA).
    \item \textbf{Biên soạn Kỹ năng và Dự án:} Hệ thống phải cho phép ứng viên thêm, sửa, xóa danh sách Kỹ năng (phân loại kỹ năng chuyên môn, kỹ năng mềm, mức độ thông thạo) và các Dự án cá nhân/thực tế đã tham gia (Tên dự án, Vị trí, Mô tả, Công nghệ sử dụng, Link demo).
    \item \textbf{Xem trước theo thời gian thực - Real-time Live Preview:} Hệ thống phải cập nhật nội dung xem trước trên giao diện mẫu CV ngay khi người dùng thao tác nhập liệu trực tiếp mà không cần tải lại trang.
    \item \textbf{Lưu và Quản lý danh sách CV:} Hệ thống phải cho phép ứng viên lưu bản nháp, tạo nhiều phiên bản CV khác nhau và quản lý danh sách CV đã tạo trong tài khoản cá nhân.
    \item \textbf{Xuất CV dạng PDF:} Hệ thống phải cho phép biên dịch dữ liệu CV thành file PDF có định dạng chuẩn, giữ nguyên kiểu dáng, phông chữ và định dạng giao diện đã chọn để người dùng tải xuống hoặc nộp trực tiếp.
    \item \textbf{Xử lý lỗi và dữ liệu nhập không hợp lệ:}
    \begin{itemize}
        \item Nếu ứng viên bỏ trống các trường bắt buộc (ví dụ: Họ tên, Email liên hệ), hệ thống phải hiển thị cảnh báo ngay tại trường dữ liệu tương ứng.
        \item Nếu quá trình xuất file PDF gặp lỗi, hệ thống phải trả về thông báo \textit{\textquotedblleft Không thể tạo file PDF, vui lòng thử lại\textquotedblright} và giữ nguyên dữ liệu đang soạn thảo.
    \end{itemize}
    \item \textbf{Tùy chỉnh bố cục:} Hệ thống phải cho phép ứng viên thay đổi thứ tự hiển thị các mục thông tin trong CV (ví dụ: Kinh nghiệm, Học vấn, Kỹ năng) bằng cách thao tác di chuyển trực tiếp và cập nhật trực quan trên canvas.
    \item \textbf{Tùy chỉnh Thiết kế và Font:} Hệ thống phải cho phép ứng viên tùy chỉnh màu sắc, font chữ và bố cục của mẫu CV đã chọn để phù hợp với phong cách cá nhân mà vẫn đảm bảo tính chuyên nghiệp.
    \item \textbf{Cấu hình Trạng thái Hiển thị CV - Public/Private:} Hệ thống phải cho phép ứng viên chủ động bật/tắt trạng thái hiển thị cho từng bản CV đã tạo giữa hai chế độ: Public (Công khai tìm việc) và Private (Riêng tư - mặc định).
\end{enumerate}

Cho phép Ứng viên tìm kiếm, lọc việc làm đa tiêu chí, thực hiện ứng tuyển trực tuyến và theo dõi trạng thái hồ sơ của mình.

\subsubsection{Các yêu cầu chức năng}

\begin{enumerate}[label=\bfseries REQ-REC-\arabic*:, leftmargin=2.5cm]
    \item \textbf{Tạo và Quản lý Tin tuyển dụng:} Hệ thống phải cho phép Nhà tuyển dụng tạo mới, cập nhật thông tin, ẩn/hiện và quản lý trạng thái hoạt động của các tin tuyển dụng, bao gồm trạng thái đang nhận hồ sơ, ngừng nhận hồ sơ và đã đóng.
    \item \textbf{Cấu hình Chi tiết Tin tuyển dụng:} Hệ thống phải bắt buộc Nhà tuyển dụng thiết lập các tiêu chí chi tiết khi tạo JD bao gồm: Tên vị trí, ngành nghề, cấp bậc (Intern, Junior, Senior,...), địa điểm làm việc, hình thức làm việc, khoảng lương (hoặc Thỏa thuận), danh sách kỹ năng yêu cầu, mô tả công việc và hạn nộp hồ sơ.
    \item \textbf{Tìm kiếm và Lọc Việc làm đa tiêu chí:} Hệ thống phải cung cấp bộ lọc cho Ứng viên tìm kiếm việc làm theo từ khóa, ngành nghề, địa điểm, mức lương, cấp bậc và hình thức làm việc.
    \item \textbf{Ứng tuyển Trực tuyến với CV SmartHire hoặc File PDF:} Hệ thống phải cho phép Ứng viên lựa chọn ứng tuyển bằng CV đã khởi tạo trực tiếp trên hệ thống hoặc tải file CV dạng PDF từ thiết bị cá nhân.
    \item \textbf{Giới hạn Đăng ký / Ứng tuyển trùng lặp:} Hệ thống phải ngăn chặn và thông báo lỗi nếu Ứng viên thực hiện nộp hồ sơ nhiều lần cho cùng một tin tuyển dụng đang mở.
    \item \textbf{Theo dõi Trạng thái Hồ sơ - Phía Ứng viên:} Hệ thống phải cung cấp cho Ứng viên giao diện xem lịch sử các công việc đã ứng tuyển kèm theo trạng thái xử lý hiện tại từ phía Nhà tuyển dụng.
    \item \textbf{Quản lý Hồ sơ Ứng tuyển - Phía Nhà tuyển dụng:} Hệ thống phải cho phép Nhà tuyển dụng xem danh sách ứng viên đã nộp hồ sơ theo từng tin tuyển dụng, mở xem nội dung CV trực tuyến và tải về.
    \item \textbf{Cập nhật Quy trình Tuyển dụng:} Hệ thống phải cho phép Nhà tuyển dụng cập nhật tiến trình hồ sơ của ứng viên theo các bước: Đã tiếp nhận $\rightarrow$ Đã duyệt / Mời phỏng vấn $\rightarrow$ Từ chối.
    \item \textbf{Xử lý lỗi và Điều kiện biên:}
    \begin{itemize}
        \item Nếu thời điểm hiện tại vượt quá hạn nộp hồ sơ do Nhà tuyển dụng cấu hình, hệ thống phải tự động chuyển tin tuyển dụng sang trạng thái hết hạn và khóa chức năng Ứng tuyển.
        \item Nếu file CV PDF ứng viên tải lên vượt quá dung lượng quy định (ví dụ: > 5MB) hoặc sai định dạng, hệ thống phải hiển thị thông báo lỗi và ngắt quá trình ứng tuyển.
    \end{itemize}
    \item \textbf{Dừng nhận hồ sơ ứng tuyển:} Hệ thống phải cho phép Nhà tuyển dụng chủ động dừng nhận hồ sơ ứng tuyển đối với một tin tuyển dụng đang mở trước thời hạn nộp hồ sơ. Khi tin tuyển dụng được chuyển sang trạng thái ngừng nhận hồ sơ, hệ thống phải ngăn Ứng viên tiếp tục nộp hồ sơ cho tin tuyển dụng đó và hiển thị thông báo phù hợp.
\end{enumerate}

    \item \textbf{Quản lý và Khóa Tài khoản Vi phạm:} Hệ thống phải cho phép Quản trị viên tìm kiếm, theo dõi danh sách tất cả người dùng (Ứng viên, Nhà tuyển dụng) và thực hiện khóa hoặc khôi phục quyền hoạt động của tài khoản.
    \item \textbf{Duyệt Tin tuyển dụng chống Spam/Lừa đảo:} Hệ thống phải cung cấp giao diện cho Quản trị viên kiểm duyệt nội dung các tin tuyển dụng trước khi hiển thị công khai trên nền tảng nhằm tránh thông tin sai sự thật hoặc gian lận.
    \item \textbf{Quản lý Danh mục Ngành nghề và Địa điểm:} Hệ thống phải cho phép Quản trị viên khởi tạo, chỉnh sửa và quản lý cấu trúc danh mục Ngành nghề chuyên môn và Tỉnh/Thành phố/Địa điểm làm việc.
    \item \textbf{Quản lý Danh mục Kỹ năng chuẩn hóa:} Hệ thống phải cho phép Quản trị viên bổ sung, chỉnh sửa danh sách Kỹ năng (Skill set) dùng chung cho toàn bộ hệ thống để phục vụ tìm kiếm và xử lý AI.
    \item \textbf{Quản lý CV Templates:} Hệ thống phải cho phép Quản trị viên quản lý danh sách các mẫu thiết kế CV (CV Templates), bật/tắt hoặc tải lên cấu trúc Template CV mới.
    \item \textbf{Thống kê và Báo cáo Vận hành:} Hệ thống phải cung cấp giao diện Dashboard báo cáo chi tiết về số lượng người dùng mới, tin tuyển dụng mở và lượt ứng tuyển thành công.
    \item \textbf{Xử lý lỗi và Ràng buộc Dữ liệu Quản trị:}

        \item Trường hợp Quản trị viên từ chối duyệt doanh nghiệp hoặc tin tuyển dụng, hệ thống bắt buộc phải nhập lý do từ chối để thông báo cho người dùng liên quan.
        \item Không cho phép xóa cứng các mục danh mục (Kỹ năng, Ngành nghề) đã có liên kết dữ liệu với CV hoặc Tin tuyển dụng đang hoạt động; hệ thống phải chuyển sang cơ chế ẩn/tắt hoạt động.
    \end{itemize}

\subsubsection{Các yêu cầu chức năng}

\begin{enumerate}[label=\bfseries REQ-AI-\arabic*:, leftmargin=2.5cm]
    \item \textbf{Trợ lý Viết CV - Gợi ý từ AI:} Hệ thống phải cung cấp tính năng gợi ý từ AI để giúp ứng viên viết CV chuyên nghiệp hơn ngay trong quá trình biên soạn.
    \item \textbf{Trợ lý Viết CV - Tối ưu hóa văn bản và Từ khóa ngành:} Hệ thống phải cho phép ứng viên gửi đoạn văn mô tả kinh nghiệm hoặc dự án để AI gợi ý viết lại chuyên nghiệp hơn và tự động bổ sung từ khóa chuẩn hóa theo ngành nghề.
    \item \textbf{Trích xuất Ngữ nghĩa Dữ liệu CV và JD:} Hệ thống phải tự động bóc tách và phân tích ngữ nghĩa (Semantic Parsing) từ nội dung CV của ứng viên và Mô tả công việc (JD) của nhà tuyển dụng.
    \item \textbf{Tính toán và Chấm điểm, phân tích Khoảng trống Kỹ năng:} Hệ thống phải tự động tính toán và trả về điểm số độ phù hợp dưới dạng tỷ lệ phần trăm (\%) giữa CV ứng viên và JD yêu cầu; đồng thời phân tích và đưa ra danh sách các kỹ năng hoặc yêu cầu mà ứng viên còn thiếu so với JD, kèm theo nhận xét chi tiết và lời khuyên cải thiện.
    \item \textbf{Gợi ý Việc làm Thông minh:} Hệ thống phải phân tích toàn bộ các bản CV của ứng viên (không phân biệt trạng thái Public hay Private) để tính toán độ tương thích và tự động gợi ý danh sách việc làm phù hợp nhất trên Dashboard cá nhân của ứng viên đó.
    \item \textbf{Xử lý Ngoại lệ và Lỗi dịch vụ AI:} Khi dịch vụ AI bị quá tải, phản hồi chậm hoặc lỗi kết nối API, hệ thống phải giữ nguyên nội dung dữ liệu hiện tại của người dùng và hiển thị thông báo \textit{\textquotedblleft Dịch vụ AI đang bận, vui lòng thử lại sau\textquotedblright}.
\end{enumerate}
