\section{Giới thiệu}

\subsection{Mục tiêu}

\textit{<Trình bày mục tiêu của tài liệu đặc tả phần mềm và những nhóm người sử dụng tài liệu này (VD: thiết kế viên, kiểm thử viên, người quản lý, v.v.).>}

\subsection{Phạm vi sản phẩm}

\textit{<Cung cấp một mô tả ngắn về phần mềm sẽ được đặc tả và mục đích của nó, bao gồm các lợi 
ích, mục tiêu liên quan (ví dụ: tên sản phẩm phần mềm là gì, phần mềm này làm cái gì). Liên hệ 
phần mềm với các mục tiêu hợp tác ha các chiến lược kinh doanh. >}

\subsection{Bảng chú giải thuật ngữ}
\textit{<Định nghĩa các từ viết tắt, các thuật ngữ được sử dụng trong tài liệu mà chúng gần như không 
được biết đến bởi người đọc. > }

\subsection{Tài liệu tham khảo}
\textit{<Liệt kê bất cứ tài liệu hay địa chỉ website nào mà bản đặc tả yêu cầu phần mềm này tham khảo 
tới. Những tài liệu tham khảo có thể là: các hướng dẫn mẫu giao diện người sử dụng, hợp đồng, 
các chuẩn, đặc tả yêu cầu hệ thống, tài liệu use-case hay tài liệu về phạm vi. Cung cấp đủ thông 
tin để người đọc có thể tìm bản sao của từng tài liệu tham khảo, bao gồm: tiêu đề, tác giả, số phát 
hành, ngày, nguồn hay nơi cung cấp. > }

\subsection{Bố cục tài liệu}
\textit{<Mô tả vắn tắt nội dung và cách tổ chức các phần còn lại của tài liệu đặc tả. Đề xuất các mục nên 
đọc với từng nhóm người đọc.>}
